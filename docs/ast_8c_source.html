116 node->left = NULL;
117 } else if (strlen(token) == 1 && ('*' <= token[0] && token[0] <= '/')) {
118 snprintf(node->value, VALUE_LEN, "%s", token);
119 node->right = stack[--stack_ptr];
120 node->left = stack[--stack_ptr];
121#if SAFE_EXP

400 if (root->left != NULL && root->right != NULL) {
401 if (is_num(root->left->value) && is_num(root->right->value)) {
402 double res = 0;
403 double left = atof(root->left->value);
404 double right = atof(root->right->value);
405 switch(root->value[0]) {
406 case '+':
407 res = left + right;
408 break;
409 case '-':
